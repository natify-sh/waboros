# waboros native code (S0016): the C reference loader (S0017) and sljit (S0016), built with CMake alone, standalone or as a
# subproject: add_subdirectory(native), or FetchContent_MakeAvailable with SOURCE_SUBDIR native.
cmake_minimum_required(VERSION 3.21)
project(waboros_native LANGUAGES C)

# Per-target C99 rules (S0017): one place, so the loader's targets and the tests agree.
function(waboros_c99 target)
  set_target_properties(${target} PROPERTIES C_STANDARD 99 C_STANDARD_REQUIRED ON C_EXTENSIONS OFF)
  target_compile_definitions(${target} PRIVATE _DEFAULT_SOURCE)
endfunction()

option(WABOROS_LOADER "build the C loader (waboros_loader) and waboros_run" ON)
option(WABOROS_LOADER_SHARED "build waboros_loader_shared (the Python shim, tests)" ${PROJECT_IS_TOP_LEVEL})
option(WABOROS_WITH_COMPILER "link the embedded compiler (D68; nothing yet)" OFF)
option(WABOROS_WITH_GC "the heap and the collector's driver (nothing yet)" ON)
option(WABOROS_GUARD "the fault handler (nothing yet)" ON)
set(WABOROS_COLLECTOR_WBO "" CACHE FILEPATH "a collector .wbo to embed (nothing yet)")
if(WABOROS_WITH_COMPILER OR PROJECT_IS_TOP_LEVEL)
  set(_sljit_default ON)
else()
  set(_sljit_default OFF)
endif()
option(WABOROS_SLJIT "build waboros_sljit (S0016) and its tests" ${_sljit_default})
option(WABOROS_SLJIT_SHARED "build a shared library (ctypes, tests); static otherwise" ${PROJECT_IS_TOP_LEVEL})
option(WABOROS_SLJIT_DEBUG "sljit's SLJIT_DEBUG and SLJIT_VERBOSE" OFF)
option(WABOROS_BUILD_TESTS "build and register the ctest self-tests" ${PROJECT_IS_TOP_LEVEL})

if(WABOROS_LOADER)
  if(MSVC)
    set(_warn /W4)
  else()
    set(_warn -Wall -Wextra -Wconversion -Wvla)
  endif()
  set(_loader_src loader/version.c loader/read.c loader/sha256.c)
  # stores, instantiation and calls (S0017 step 2): POSIX for now; Windows is step 8 (os_windows.c)
  if(NOT WIN32)
    list(APPEND _loader_src loader/os_posix.c loader/entry.c loader/instance.c)
  endif()
  add_library(waboros_loader STATIC ${_loader_src})
  add_library(waboros::loader ALIAS waboros_loader)
  set(_loader_targets waboros_loader)
  if(WABOROS_LOADER_SHARED)
    add_library(waboros_loader_shared SHARED ${_loader_src})
    set_target_properties(waboros_loader_shared PROPERTIES WINDOWS_EXPORT_ALL_SYMBOLS ON)
    list(APPEND _loader_targets waboros_loader_shared)
  endif()
  foreach(_t ${_loader_targets})
    waboros_c99(${_t})
    set_target_properties(${_t} PROPERTIES POSITION_INDEPENDENT_CODE ON)
    target_include_directories(${_t} PUBLIC "$<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/loader>")
    target_compile_options(${_t} PRIVATE ${_warn})
    if(NOT WIN32)
      target_compile_definitions(${_t} PRIVATE WABOROS_RUNTIME=1)
      find_package(Threads REQUIRED)
      target_link_libraries(${_t} PUBLIC Threads::Threads)
    endif()
  endforeach()
  add_executable(waboros_run tools/waboros_run.c)
  target_link_libraries(waboros_run PRIVATE waboros_loader)
  waboros_c99(waboros_run)
  target_compile_options(waboros_run PRIVATE ${_warn})
endif()

if(WABOROS_SLJIT)
  set(sljit_dir "${CMAKE_CURRENT_SOURCE_DIR}/third_party/sljit/sljit_src")
  if(NOT EXISTS "${sljit_dir}/sljitLir.c")
    message(FATAL_ERROR "sljit is missing: git submodule update --init native/third_party/sljit")
  endif()

  # One translation unit (it includes sljitLir.c); sljit's own CMakeLists.txt is not used.
  if(WABOROS_SLJIT_SHARED)
    set(_kind SHARED)
  else()
    set(_kind STATIC)
  endif()
  add_library(waboros_sljit ${_kind} "${CMAKE_CURRENT_SOURCE_DIR}/sljit/waboros_sljit.c")
  add_library(waboros::sljit ALIAS waboros_sljit)
  set_target_properties(waboros_sljit PROPERTIES
    C_STANDARD 99 C_STANDARD_REQUIRED ON C_EXTENSIONS OFF POSITION_INDEPENDENT_CODE ON
    WINDOWS_EXPORT_ALL_SYMBOLS ON)  # the DLL exports sljit's API and waboros_sljit_* (no __declspec in sljit)
  target_include_directories(waboros_sljit PUBLIC
    "$<BUILD_INTERFACE:${sljit_dir}>" "$<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/sljit>")
  target_compile_definitions(waboros_sljit PRIVATE _DEFAULT_SOURCE)
  target_compile_definitions(waboros_sljit PUBLIC SLJIT_CONFIG_AUTO=1)
  # sljit picks LSE atomics from __ARM_FEATURE_ATOMICS; Apple Clang's arm64 default (apple-m1) has it,
  # the portable baseline is ARMv8.0-A (S0016). Other OSes default to the generic ARMv8.0-A already.
  if(APPLE AND CMAKE_SYSTEM_PROCESSOR MATCHES "^(arm64|aarch64)$")
    target_compile_options(waboros_sljit PRIVATE -march=armv8-a)
  endif()
  # DEBUG and VERBOSE change the layout of struct sljit_compiler, so they are PUBLIC (users see the same)
  if(WABOROS_SLJIT_DEBUG)
    target_compile_definitions(waboros_sljit PUBLIC SLJIT_VERBOSE=1 SLJIT_DEBUG=1)
  else()
    target_compile_definitions(waboros_sljit PUBLIC SLJIT_VERBOSE=0 SLJIT_DEBUG=0)
  endif()
  if(NOT WABOROS_SLJIT_SHARED)  # sljit's symbols stay internal, only waboros_sljit_* is visible
    target_compile_definitions(waboros_sljit PUBLIC SLJIT_CONFIG_STATIC=1)
  endif()
endif()

if(WABOROS_BUILD_TESTS)
  enable_testing()
  get_property(_multi GLOBAL PROPERTY GENERATOR_IS_MULTI_CONFIG)
  set(_cfg "")
  set(_ccfg "")
  if(_multi)
    set(_cfg --build-config $<CONFIG>)
    set(_ccfg -C $<CONFIG>)
  endif()
  if(WABOROS_LOADER)
    add_executable(waboros_test_smoke tests/loader/smoke.c)
    target_link_libraries(waboros_test_smoke PRIVATE waboros::loader)
    waboros_c99(waboros_test_smoke)
    add_test(NAME loader_smoke COMMAND waboros_test_smoke)
    add_test(NAME loader_run_version COMMAND waboros_run --version)
    # S0017 step 2: the runtime's entry and sp_stub run hand-assembled C entries (x86-64 SysV hosts)
    if(NOT WIN32 AND CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|AMD64|amd64)$")
      add_executable(waboros_test_entry tests/loader/test_entry.c)
      target_link_libraries(waboros_test_entry PRIVATE waboros::loader)
      target_include_directories(waboros_test_entry PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/loader")
      waboros_c99(waboros_test_entry)
      target_compile_options(waboros_test_entry PRIVATE ${_warn})
      add_test(NAME loader_entry COMMAND waboros_test_entry)
    endif()
    # S0017 step 1: the reader on the committed files (tests/artifact/data, absent from a copy of native/ alone).
    # The host's explicit files are accepted; the other architecture's are refused at step 5 (thin) or need a
    # compiler (fat). Guard files are the handler's (step 4).
    if(CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|AMD64|amd64)$")
      set(_arch x86_64)
      set(_other aarch64)
    elseif(CMAKE_SYSTEM_PROCESSOR MATCHES "^(aarch64|arm64|ARM64)$")
      set(_arch aarch64)
      set(_other x86_64)
    endif()
    if(_arch)
      set(_data "${CMAKE_CURRENT_SOURCE_DIR}/../tests/artifact/data")
      file(GLOB _own "${_data}/v1-d75-*-${_arch}-explicit*.wbo")
      file(GLOB _foreign "${_data}/v1-d75-*-${_other}-explicit*.wbo")
      if(_own AND _foreign)
        add_test(NAME loader_check_committed COMMAND waboros_run --check ${_own})
        set_tests_properties(loader_check_committed PROPERTIES PASS_REGULAR_EXPRESSION "accepted"
          FAIL_REGULAR_EXPRESSION "refused;needs a compiler")
        add_test(NAME loader_check_other_architecture COMMAND waboros_run --check ${_foreign})
        set_tests_properties(loader_check_other_architecture PROPERTIES PASS_REGULAR_EXPRESSION "refused check=5"
          FAIL_REGULAR_EXPRESSION "accepted")
      endif()
    endif()
  endif()
  if(WABOROS_SLJIT)
    if(NOT WABOROS_SLJIT_SHARED)
      message(FATAL_ERROR "the tests call sljit's API: WABOROS_BUILD_TESTS needs WABOROS_SLJIT_SHARED")
    endif()
    add_executable(waboros_test_native tests/test_native.c)
    target_link_libraries(waboros_test_native PRIVATE waboros_sljit)
    waboros_c99(waboros_test_native)
    add_test(NAME native COMMAND waboros_test_native)
    if(CMAKE_SYSTEM_PROCESSOR MATCHES "^(aarch64|arm64|ARM64)$" AND NOT WIN32)
      add_test(NAME bti_pingpong COMMAND waboros_test_native bti 1)
      if(CMAKE_CROSSCOMPILING_EMULATOR)  # the control: QEMU enforces BTI, so without the pad it is SIGILL
        add_test(NAME bti_control COMMAND waboros_test_native bti 0)
      endif()
    endif()
  endif()
  # the embedding check: a tiny project that takes native/ as a subproject, in three ways
  set(_tc "")
  if(CMAKE_CROSSCOMPILING)
    string(REPLACE ";" "\;" _emu "${CMAKE_CROSSCOMPILING_EMULATOR}")
    set(_tc -DCMAKE_SYSTEM_NAME=${CMAKE_SYSTEM_NAME} -DCMAKE_SYSTEM_PROCESSOR=${CMAKE_SYSTEM_PROCESSOR}
      -DCMAKE_C_COMPILER=${CMAKE_C_COMPILER} "-DCMAKE_CROSSCOMPILING_EMULATOR=${_emu}")
  endif()
  # name:embedding:sljit; the embedded default is no sljit, so the first two ask for it
  set(_consumers add_subdirectory:add_subdirectory:ON FetchContent:FetchContent:ON nosljit:add_subdirectory:OFF)
  if(NOT WABOROS_SLJIT)
    list(FILTER _consumers EXCLUDE REGEX ":ON$")
  endif()
  foreach(_c ${_consumers})
    string(REPLACE ":" ";" _p ${_c})
    list(GET _p 0 _name)
    list(GET _p 1 _how)
    list(GET _p 2 _sljit)
    add_test(NAME consumer_${_name} COMMAND ${CMAKE_CTEST_COMMAND} --build-and-test
      "${CMAKE_CURRENT_SOURCE_DIR}/tests/consumer" "${CMAKE_CURRENT_BINARY_DIR}/consumer_${_name}"
      --build-generator "${CMAKE_GENERATOR}" ${_cfg}
      --build-options --warn-uninitialized -Werror=dev "-DWABOROS_ROOT=${CMAKE_CURRENT_SOURCE_DIR}/.."
      -DEMBED=${_how} -DWABOROS_SLJIT=${_sljit} ${_tc}
      --test-command ${CMAKE_CTEST_COMMAND} ${_ccfg} --output-on-failure)
  endforeach()
endif()
